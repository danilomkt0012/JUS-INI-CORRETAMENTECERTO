"""

import os
import requests
import logging
from datetime import datetime
from typing import Dict, Any
class AllPayAPI:
    def __init__(self):
        self.base_url = "https://api.allpaygateway.com.br/api/public/v1"
        self.api_token = os.getenv("ALLPAY_API_TOKEN", "KmDzUymcpZI9bGSJKVjTdxS9GAZurxy1lj7PXcx3N7s1g5yVwugLivabPLFT")
        self.offer_hash = os.getenv("ALLPAY_OFFER_HASH", "jqbqtusndc")
        self.product_hash = os.getenv("ALLPAY_PRODUCT_HASH", "yt1etnzg8z")
        self.logger = logging.getLogger(__name__)
        
        self.logger.info("✅ AllPay API inicializada")
    
    def create_pix_payment(self, customer_data: Dict[str, Any], amount_reais: float = 117.35) -> Dict[str, Any]:
        """
        Cria pagamento PIX via AllPay
        
        Args:
            customer_data: {'name': 'Nome', 'email': 'email@teste.com', 'cpf': '123.456.789-10'}
            amount_reais: Valor em reais
            
        Returns:
            {'success': True/False, 'transaction_id': 'xxx', 'pix_key': 'codigo_pix'}
        """
        try:
            url = f"{self.base_url}/transactions"
            amount_centavos = int(amount_reais * 100)
            
            # Dados da transação AllPay
            payment_data = {
                "amount": amount_centavos,
                "offer_hash": self.offer_hash,
                "payment_method": "pix",
                "customer": {
                    "name": customer_data.get('name', 'Cliente'),
                    "email": customer_data.get('email', 'cliente@email.com'),
                    "phone_number": "11987654321",
                    "document": customer_data.get('cpf', '').replace('.', '').replace('-', ''),
                    "street_name": "Rua Principal",
                    "number": "123",
                    "complement": "",
                    "neighborhood": "Centro", 
                    "city": "São Paulo",
                    "state": "SP",
                    "zip_code": "01000000"
                },
                "cart": [
                    {
                        "product_hash": self.product_hash,
                        "title": "Produto Digital",
                        "price": amount_centavos,
                        "quantity": 1,
                        "operation_type": 1,
                        "tangible": False,
                        "product_id": 1,
                        "offer_id": 1
                    }
                ],
                "installments": 1,
                "expire_in_days": 1,
                "transaction_origin": "api",
                "tracking": {
                    "src": "",
                    "utm_source": "",
                    "utm_medium": "",
                    "utm_campaign": "",
                    "utm_term": "",
                    "utm_content": ""
                }
            }
            
            # Fazer requisição
            response = requests.post(
                f"{url}?api_token={self.api_token}",
                json=payment_data,
                headers={
                    'Content-Type': 'application/json',
                    'Accept': 'application/json',
                    'User-Agent': 'AllPay-Replit/1.0'
                },
                timeout=10
            )
            
            if response.status_code in [200, 201]:
                result = response.json()
                transaction_id = result.get('hash') or str(result.get('id', ''))
                
                # Buscar PIX nos dados
                pix_data = result.get('pix', {})
                pix_key = pix_data.get('pix_qr_code') or pix_data.get('pix_url')
                
                payment_status = result.get('payment_status', 'pending')
                
                if transaction_id and pix_key:
                    self.logger.info(f"✅ AllPay PIX criado - ID: {transaction_id}")
                    return {
                        'success': True,
                        'transaction_id': transaction_id,
                        'pix_key': pix_key,
                        'qr_code': pix_key,
                        'amount': amount_reais,
                        'status': payment_status,
                        'provider': 'AllPay',
                        'created_at': datetime.now().isoformat()
                    }
                else:
                    self.logger.warning(f"⚠️ AllPay: transação criada mas sem PIX")
                    return {'success': False, 'error': f'AllPay sem PIX - Status: {payment_status}'}
            else:
                error_msg = f"HTTP {response.status_code} - {response.text}"
                self.logger.error(f"❌ AllPay erro: {error_msg}")
                return {'success': False, 'error': error_msg}
                
        except Exception as e:
            self.logger.error(f"💥 AllPay exception: {str(e)}")
            return {'success': False, 'error': f'Erro: {str(e)}'}
    
    def check_payment_status(self, transaction_id: str) -> Dict[str, Any]:
        """
        Verifica status do pagamento
        """
        try:
            url = f"{self.base_url}/transactions/{transaction_id}"
            
            response = requests.get(
                f"{url}?api_token={self.api_token}",
                headers={'User-Agent': 'AllPay-Replit/1.0'},
                timeout=5
            )
            
            if response.status_code == 200:
                result = response.json()
                status = result.get('status', 'pending').lower()
                paid = status in ['paid', 'approved', 'completed']
                
                return {
                    'success': True,
                    'status': 'paid' if paid else 'pending',
                    'paid': paid,
                    'pending': not paid,
                    'amount': result.get('amount', 0) / 100,
                    'provider': 'AllPay'
                }
            else:
                return {
                    'success': True,
                    'status': 'pending',
                    'paid': False,
                    'pending': True,
                    'provider': 'AllPay'
                }
                
        except Exception as e:
            return {'success': False, 'error': str(e)}

Flask App com AllPay PIX - Pronto para Replit

import logging
from allpay_api import AllPayAPI
# Configurar logging
logging.basicConfig(level=logging.INFO)
app = Flask(__name__)
app.secret_key = os.environ.get("SESSION_SECRET", "dev-secret-change-me")
# Inicializar AllPay
allpay = AllPayAPI()
@app.route('/')
def home():
    """Página principal com formulário PIX"""
    return render_template_string(FRONTEND_HTML)
@app.route('/api/generate-pix', methods=['POST'])
def generate_pix():
    """API para gerar PIX"""
    try:
        data = request.get_json()
        
        customer_data = {
            'name': data.get('name', ''),
            'email': data.get('email', ''),
            'cpf': data.get('cpf', '')
        }
        
        amount = float(data.get('amount', 117.35))
        
        # Validar dados
        if not all([customer_data['name'], customer_data['email'], customer_data['cpf']]):
            return jsonify({
                'success': False,
                'error': 'Nome, email e CPF são obrigatórios'
            }), 400
        
        # Gerar PIX
        result = allpay.create_pix_payment(customer_data, amount)
        
        if result['success']:
            return jsonify({
                'success': True,
                'transaction_id': result['transaction_id'],
                'pix_code': result['pix_key'],
                'qr_code': result['pix_key'], 
                'amount': result['amount'],
                'provider': result['provider'],
                'status': result.get('status', 'pending')
            })
        else:
            return jsonify({
                'success': False,
                'error': result['error']
            }), 500
            
    except Exception as e:
        return jsonify({
            'success': False,
            'error': f'Erro interno: {str(e)}'
        }), 500
@app.route('/api/check-payment/<transaction_id>')
def check_payment(transaction_id):
    """API para verificar status"""
    try:
        result = allpay.check_payment_status(transaction_id)
        return jsonify(result)
        
    except Exception as e:
        return jsonify({
            'success': False,
            'error': str(e)
        }), 500
